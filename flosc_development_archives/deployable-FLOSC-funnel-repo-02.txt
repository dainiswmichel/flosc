Below is a complete, deployable FLOSC funnel repo (FastAPI + vanilla HTML/JS) that:

Loads all creator content/pricing/logic from two CSVs

Records audio in-browser (MediaRecorder), uploads it

Transcribes locally with self-hosted faster-whisper

Compares transcript vs expected sentence using phoneme-level comparison (CMU phonemes) and maps to phoneme_groups

Forces Supabase magic-link login to view results

Sends Brevo email sequence (immediate + 12h + 23h) with personalized sound gaps + 1–2 targeted free lessons + 24h OTO

Uses Stripe Checkout + webhook to mark paid

Serves a protected dashboard with free lessons (post-offer) + full lessons (paid)

This is built for a single DigitalOcean droplet with HTTPS, Nginx reverse proxy, and systemd services.

Magic sentences (also CSV-driven)

To keep “two CSVs only”, we store magic sentences inside targeted_lessons.csv as special rows, OR you can add a second file magic_sentences.csv.
But you explicitly want two CSVs. So we do this:

Add rows where lesson_id starts with ms_ and content_path holds the sentence text inline (or a path). Example:

lesson_id,title,phoneme_group,free_rank,content_path,is_free
ms_set_a_1,Magic Sentence 1,MAGIC_SENTENCE,0,"The ship is in the big city.",false
ms_set_a_2,Magic Sentence 2,MAGIC_SENTENCE,0,"Pack the black bag.",false
ms_set_a_3,Magic Sentence 3,MAGIC_SENTENCE,0,"Dad had an apple.",false
l001,Fix /ɪ/ vs /iː/,VOWEL_I,1,lessons/l001.html,true


Backend filters phoneme_group == MAGIC_SENTENCE and lesson_id prefix ms_set_a_.

DB_PATH = os.path.join(settings.data_dir, "flosc.sqlite")

def conn():
    os.makedirs(settings.data_dir, exist_ok=True)
    c = sqlite3.connect(DB_PATH, check_same_thread=False)

def init_db():
    c = conn()
    cur = c.cursor()
    cur.execute("""
    CREATE TABLE IF NOT EXISTS quizzes (
        quiz_id TEXT PRIMARY KEY,
        created_at INTEGER,
        expected_sentence TEXT,
        email_optional TEXT,
        audio_path TEXT,
        transcript TEXT,
        mismatched_groups TEXT,
        mismatched_phonemes TEXT,
        score REAL,
        user_id TEXT,
        offer_sent_at INTEGER,
        oto_expires_at INTEGER,
        paid INTEGER DEFAULT 0
    )
    """)
    cur.execute("""
    CREATE TABLE IF NOT EXISTS events (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        quiz_id TEXT,
        ts INTEGER,
        type TEXT,
        payload TEXT
    )
    """)
    c.commit()
    c.close()

backend/app/csv_loader.py
import csv, os
from .config import settings

PRODUCT_CSV = os.path.join(os.path.dirname(__file__), "config_data", "product.csv")
TARGET_CSV = os.path.join(os.path.dirname(__file__), "config_data", "targeted_lessons.csv")

def load_product(product_id: str = None) -> dict:
    with open(PRODUCT_CSV, newline="", encoding="utf-8") as f:
        rows = list(csv.DictReader(f))
    if not rows:
        raise RuntimeError("product.csv is empty")
    if product_id:
        for r in rows:
            if r["product_id"] == product_id:
                return r
        raise RuntimeError(f"product_id {product_id} not found")
    return rows[0]

def load_target_rows() -> list[dict]:
    with open(TARGET_CSV, newline="", encoding="utf-8") as f:
        return list(csv.DictReader(f))

def get_magic_sentences(set_prefix: str) -> list[str]:
    rows = load_target_rows()
    ms = []
    for r in rows:
        if r.get("phoneme_group") == "MAGIC_SENTENCE" and r.get("lesson_id","").startswith(f"ms_{set_prefix}_"):
            ms.append(r.get("content_path","").strip().strip('"'))
    return ms[:3]

def select_lessons_for_groups(groups: list[str], limit: int = 2) -> list[dict]:
    rows = load_target_rows()
    candidates = []
    for r in rows:
        if r.get("phoneme_group") in groups and str(r.get("is_free","")).lower() == "true":
            candidates.append(r)
    candidates.sort(key=lambda x: int(x.get("free_rank") or 9999))
    return candidates[:limit]

def all_lessons(is_free: bool) -> list[dict]:
    rows = load_target_rows()
    out = []
    for r in rows:
        if r.get("phoneme_group") == "MAGIC_SENTENCE":
            continue
        flag = str(r.get("is_free","")).lower() == "true"
        if flag == is_free:
            out.append(r)
    out.sort(key=lambda x: int(x.get("free_rank") or 9999))
    return out

backend/app/whisper_service.py
from faster_whisper import WhisperModel

# tiny or base; base is slower but more accurate
MODEL_NAME = "tiny"

_model = None

def model():
    global _model
    if _model is None:
        _model = WhisperModel(MODEL_NAME, device="cpu", compute_type="int8")
    return _model

def transcribe(audio_path: str) -> str:
    segments, info = model().transcribe(audio_path, beam_size=5)
    text = " ".join([s.text.strip() for s in segments]).strip()
    return text

backend/app/phoneme.py (CMU phonemes + group mapping)
import re
import pronouncing

# Map CMU phoneme prefixes to your groups
PHONEME_GROUPS = {
    # Vowels (examples)
    "AE": "VOWEL_AE",   # /æ/
    "IH": "VOWEL_I",    # /ɪ/
    "IY": "VOWEL_I_LONG", # /iː/
    "EH": "VOWEL_E",
    "AH": "VOWEL_AH",
    "AA": "VOWEL_AA",
    "UH": "VOWEL_UH",
    "UW": "VOWEL_UW",
    "OW": "VOWEL_OW",
    "ER": "VOWEL_ER",

    # Consonant clusters (optional)
    "TH": "TH_SOUND",
    "DH": "TH_SOUND",
    "R":  "R_SOUND",
    "L":  "L_SOUND",
    "W":  "W_SOUND",
}

def normalize_text(s: str) -> str:
    s = s.lower()
    s = re.sub(r"[^a-z\s']", " ", s)
    s = re.sub(r"\s+", " ", s).strip()
    return s

def cmu_phonemes_for_word(word: str) -> list[str]:
    phones = pronouncing.phones_for_word(word.lower())
    if not phones:
        return []
    # take the first pronunciation
    return phones[0].split()

def cmu_phonemes_for_sentence(sentence: str) -> list[str]:
    sent = normalize_text(sentence)
    out = []
    for w in sent.split():
        out.extend(cmu_phonemes_for_word(w))
    return out

def strip_stress(ph: str) -> str:
    return re.sub(r"\d", "", ph)

def phoneme_groups(phonemes: list[str]) -> list[str]:
    groups = []
    for ph in phonemes:
        base = strip_stress(ph)
        g = PHONEME_GROUPS.get(base)
        if g:
            groups.append(g)
    return groups

def compare_expected_vs_transcript(expected: str, transcript: str) -> dict:
    exp = cmu_phonemes_for_sentence(expected)
    got = cmu_phonemes_for_sentence(transcript)

    # Simple overlap score + diff list (lightweight, robust enough for MVP)
    exp_base = [strip_stress(p) for p in exp]
    got_base = [strip_stress(p) for p in got]

    exp_set = set(exp_base)
    got_set = set(got_base)

    missing = sorted(list(exp_set - got_set))
    extra = sorted(list(got_set - exp_set))

    mismatched = missing  # treat missing as the “needs work” set for sounds
    mismatched_groups = sorted(list(set([PHONEME_GROUPS.get(p) for p in mismatched if PHONEME_GROUPS.get(p)])))

    score = 0.0
    if exp_set:
        score = len(exp_set & got_set) / len(exp_set)

    return {
        "expected_phonemes": exp_base,
        "got_phonemes": got_base,
        "mismatched_phonemes": mismatched,
        "mismatched_groups": mismatched_groups,
        "extra_phonemes": extra,
        "score": round(score, 3),
    }

backend/app/auth.py (verify Supabase JWT via JWKS)
import time, json
import httpx
import jwt
from jwt import PyJWKClient
from .config import settings

_jwk_client = None

def jwk_client():
    global _jwk_client
    if _jwk_client is None:
        _jwk_client = PyJWKClient(settings.supabase_jwks_url)
    return _jwk_client

def verify_supabase_jwt(token: str) -> dict:
    signing_key = jwk_client().get_signing_key_from_jwt(token).key
    payload = jwt.decode(
        token,
        signing_key,
        algorithms=["RS256"],
        options={"verify_aud": False},
    )
    # Basic expiry check is already in jwt.decode
    return payload

import httpx
from .config import settings

BREVO_URL = "https://api.brevo.com/v3/smtp/email"

async def send_email(to_email: str, subject: str, html_content: str, from_email: str, from_name: str):
    headers = {
        "api-key": settings.brevo_api_key,
        "content-type": "application/json",
        "accept": "application/json",
    }
    payload = {
        "sender": {"email": from_email, "name": from_name},
        "to": [{"email": to_email}],
        "subject": subject,
        "htmlContent": html_content,
    }
    async with httpx.AsyncClient(timeout=20) as client:
        r = await client.post(BREVO_URL, headers=headers, json=payload)
        r.raise_for_status()
        return r.json()

        metadata={"quiz_id": quiz_id},
    )
    return session.url

async def _send(template_name: str, quiz_id: str, subject: str):
    c = conn()
    row = c.execute("SELECT * FROM quizzes WHERE quiz_id=?", (quiz_id,)).fetchone()
    if not row or not row["email_optional"]:

    product = load_product()

        "currency": product["currency"],
        "checkout_url": f"{product['stripe_success_url']}?quiz_id={quiz_id}"
    })
    await send_email(
        row["email_optional"],
        subject,
        html,
        product["from_email"],
        product["from_name"],
    )

@app.post("/api/quiz/create")
def quiz_create(product_id: str = Form(default="")):
    product = load_product(product_id or None)
    quiz_id = new_quiz_id()
    ms = get_magic_sentences(product["magic_sentence_set_id"])
    if len(ms) < 3:
        raise HTTPException(500, "Need 3 magic sentences in targeted_lessons.csv")

    )
    c.commit()
    c.close()

    return {"quiz_id": quiz_id, "magic_sentences": ms}

@app.post("/api/quiz/upload-audio")
async def quiz_upload_audio(
    quiz_id: str = Form(...),
    expected_sentence: str = Form(...),
    email_optional: str = Form(default=""),
    audio: UploadFile = File(...),
):
    path = audio_path_for_quiz(quiz_id)
    with open(path, "wb") as f:
        f.write(await audio.read())

    # process immediately (MVP). For higher scale, push to worker.
    transcript_text = transcribe(path)
    cmp = compare_expected_vs_transcript(expected_sentence, transcript_text)

    c = conn()
    c.execute("""
        UPDATE quizzes
        SET expected_sentence=?, email_optional=?, audio_path=?, transcript=?,
            mismatched_groups=?, mismatched_phonemes=?, score=?
        WHERE quiz_id=?
    """, (
        expected_sentence,
        email_optional.strip() or None,
        path,
        transcript_text,
        json.dumps(cmp["mismatched_groups"]),
        json.dumps(cmp["mismatched_phonemes"]),
        float(cmp["score"]),
        quiz_id
    ))
    c.commit()
    c.close()

    return {
        "quiz_id": quiz_id,
        "expected_sentence": expected_sentence,
        "transcript": transcript_text,
        "mismatched_groups": cmp["mismatched_groups"],
        "mismatched_phonemes": cmp["mismatched_phonemes"],
        "score": cmp["score"],
    }

@app.post("/api/quiz/attach-user")
def quiz_attach_user(
    quiz_id: str,
    authorization: str = Header(default="")
):
    if not authorization.startswith("Bearer "):
        raise HTTPException(401, "Missing bearer token")
    token = authorization.split(" ", 1)[1]
    payload = verify_supabase_jwt(token)
    user_id = payload.get("sub")
    if not user_id:
        raise HTTPException(401, "Bad token")

    c = conn()
    c.execute("UPDATE quizzes SET user_id=? WHERE quiz_id=?", (user_id, quiz_id))
    c.commit()
    c.close()
    return {"ok": True, "user_id": user_id}

@app.get("/api/offer/preview")
def offer_preview(
    quiz_id: str,
    authorization: str = Header(default="")
):
    if not authorization.startswith("Bearer "):
        raise HTTPException(401, "Missing bearer token")
    token = authorization.split(" ", 1)[1]
    payload = verify_supabase_jwt(token)
    user_id = payload.get("sub")

    c = conn()
    row = c.execute("SELECT * FROM quizzes WHERE quiz_id=?", (quiz_id,)).fetchone()
    c.close()
    if not row or row["user_id"] != user_id:
        raise HTTPException(403, "Not your quiz")

    product = load_product()
    groups = json.loads(row["mismatched_groups"] or "[]")
    lessons = select_lessons_for_groups(groups, limit=2)

    # set OTO expiry if not set
    oto_hours = int(product["oto_hours"])
    now = int(time.time())
    oto_expires = row["oto_expires_at"] or (now + oto_hours * 3600)
    if not row["oto_expires_at"]:
        c = conn()
        c.execute("UPDATE quizzes SET oto_expires_at=? WHERE quiz_id=?", (oto_expires, quiz_id))
        c.commit()
        c.close()

    return {
        "quiz_id": quiz_id,
        "sound_count": len(groups),
        "sound_list": groups,
        "free_lessons": lessons,
        "oto_price_cents": int(product["oto_price_cents"]),
        "full_price_cents": int(product["full_price_cents"]),
        "currency": product["currency"],
        "oto_expires_at": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime(oto_expires)),
    }

    user_email = payload.get("email")

    c = conn()
    row = c.execute("SELECT * FROM quizzes WHERE quiz_id=?", (quiz_id,)).fetchone()
    c.close()
    if not row or row["user_id"] != user_id:
        raise HTTPException(403, "Not your quiz")

    # OTO expiry enforcement
    now = int(time.time())
    if row["oto_expires_at"] and now > int(row["oto_expires_at"]):
        raise HTTPException(403, "Offer expired")

    if not user_email:
        raise HTTPException(400, "No email on Supabase user; add email to Supabase auth")

    product = load_product()
    url = create_checkout_session(quiz_id, product, user_email)
    return {"checkout_url": url}

@app.post("/api/stripe/webhook")
async def stripe_webhook(request: Request):
    import stripe
    from .config import settings
    payload = await request.body()
    sig = request.headers.get("stripe-signature","")
    try:
        event = stripe.Webhook.construct_event(payload, sig, settings.stripe_webhook_secret)
    except Exception:
        raise HTTPException(400, "Invalid webhook")

    if event["type"] == "checkout.session.completed":
        sess = event["data"]["object"]
        quiz_id = (sess.get("metadata") or {}).get("quiz_id")
        if quiz_id:
            c = conn()
            c.execute("UPDATE quizzes SET paid=1 WHERE quiz_id=?", (quiz_id,))
            c.commit()
            c.close()

    return {"received": True}

async function api(path, opts = {}) {
  const res = await fetch(`${API}${path}`, opts);

export async function createQuiz() {

export async function uploadAudio({quiz_id, expected_sentence, email_optional, blob}) {

}

Security + anonymized audio storage (what this MVP does)

Audio is stored as /opt/flosc/audio/{quiz_id}.webm (UUID-only naming).

No public endpoint serves audio.

Quiz is “anonymous” until Supabase login attaches user_id.

Block logic (3 “No” → blocked) is client-side in this MVP; for stronger enforcement you’d persist strikes by IP + cookie hash server-side.

Notes on phoneme accuracy (honest constraints)

This MVP compares expected vs transcript via CMU phonemes (English pronunciation dictionary).

If Whisper outputs a different word than expected, phoneme differences will follow. That’s acceptable for “sound gaps” coaching, but not a forensic phoneme aligner.

If you want true phoneme alignment from audio (not transcript), that’s a bigger step (forced alignment models). This build stays lightweight and droplet-friendly.

Personalization: “Work needed on these sounds: {count} — {list}”

Dynamic free lessons: pulled from CSV by phoneme_group

24h urgency: enforced by backend with oto_expires_at

Automatic reminders: RQ scheduled at 12h and 23h

```php
 * FLOSC State Object
 * Single source of truth for user, flow, quiz, and conversational context
 *
 * @package FLOSC
 * @since 1.9.0

class FLOSC_State {
    private $user_id;
    private $flow_id;
    private $state;

    public function __construct($user_id, $flow_id) {
        $this->user_id = $user_id;
        $this->flow_id = $flow_id;
        $this->state = $this->flosc_build_state();
    }

    private function flosc_build_state() {
        // Leverage existing Condition Evaluator - don't rebuild what works
        $context = FLOSC_Condition_Evaluator::build_context($this->user_id, $this->flow_id);

        // Enhance with flow-specific data
        $flow = flosc_flows()->get_current_flow();

        return [
            // Identity & Access
            'user_id' => $this->user_id,
            'flow_id' => $this->flow_id,
            'phase' => $context['phase'] ?? 'freeline',
            'access_level' => $this->flosc_determine_access_level($context),
            'user_type' => $this->flosc_determine_user_type($context), // admin/visitor/guest/member

            // Flow Configuration
            'flow' => [
                'name' => $flow['name'] ?? 'Unknown',
                'slug' => $flow['slug'] ?? '',
                'wp_category_id' => $flow['wp_category_id'] ?? 0,
                'ivr_file' => $flow['ivr_file'] ?? 'flosc_default_ivr.md',
                'custom_domain' => $flow['custom_domain'] ?? '',
            ],

            // Quiz State
            'quiz' => [
                'taken' => $context['quiz_taken'] ?? false,
                'score' => $context['score'] ?? 0,
                'results_shown' => $context['quiz_results_shown'] ?? false,
                'free_lesson_number' => $context['free_lesson_number'] ?? null,
                'free_lesson_assigned' => $context['free_lesson_assigned'] ?? false,
                'free_lesson_viewed' => $context['free_lesson_viewed'] ?? false,
            ],

            // IVR Context
            'ivr' => [
                'active_conditions' => $context['active_conditions'] ?? [],
                'visible_autoprompts' => $this->flosc_get_visible_autoprompts($context),
                'boundary_rules' => $this->flosc_get_boundary_rules($context),
            ],

            // Learning Progress
            'progress' => [
                'lessons_completed' => $context['lessons_completed'] ?? 0,
                'current_lesson' => $context['current_lesson'] ?? null,
                'last_activity' => $context['last_activity'] ?? null,
            ],

            // Session
            'session_id' => $context['session_id'] ?? null,
            'visitor_id' => $context['visitor_id'] ?? null,
        ];
    }

    public function flosc_get($key = null) {
        if ($key === null) return $this->state;
        return $this->state[$key] ?? null;
    }

    public function flosc_get_state_hash() {
        // For cache keying: hash only state that affects AI responses
        $cache_relevant = [
            'phase' => $this->state['phase'],
            'access_level' => $this->state['access_level'],
            'quiz_taken' => $this->state['quiz']['taken'],
            'free_lesson_number' => $this->state['quiz']['free_lesson_number'],
        ];
        return md5(json_encode($cache_relevant));
    }

    private function flosc_determine_user_type($context) {
        if (current_user_can('manage_options')) return 'admin';
        if ($context['access_level'] === 'member') return 'member';
        if ($context['quiz_taken']) return 'guest';
        return 'visitor';
    }

    private function flosc_determine_access_level($context) {
        if ($context['purchased'] ?? false) return 'member';
        if ($context['quiz_taken'] ?? false) return 'guest';
        if (is_user_logged_in()) return 'user';
        return 'visitor';
    }

    private function flosc_get_visible_autoprompts($context) {
        // Parse IVR and extract visible autoprompts for current phase
        $ivr_parser = new FLOSC_IVR_Parser($this->state['flow']['ivr_file']);
        return $ivr_parser->get_visible_autoprompts($this->state['phase'], $context);
    }

    private function flosc_get_boundary_rules($context) {
        $user_type = $this->flosc_determine_user_type($context);

        $rules = [
            'admin' => [
                'can_see_config' => true,
                'can_see_all_lessons' => true,
                'can_see_pricing' => true,
                'can_debug' => true,
            ],
            'member' => [
                'can_see_all_lessons' => true,
                'can_see_pricing' => false,
                'can_see_catalog' => true,
            ],
            'guest' => [
                'can_see_free_lesson' => true,
                'can_see_catalog' => true,
                'can_see_pricing' => true,
                'must_encourage_purchase' => true,
            ],
            'visitor' => [
                'can_see_catalog' => true,
                'must_encourage_quiz' => true,
                'can_see_pricing' => false,
            ],
        ];

        return $rules[$user_type] ?? $rules['visitor'];
    }
}
```

```php
<?php
/**
 * FLOSC Tool Guardian
 * Server-side enforcement layer - validates tool access BEFORE execution
 * Deny-by-default philosophy: only explicitly allowed actions proceed
 *
 * @package FLOSC
 * @since 1.9.0
 */
class FLOSC_Tool_Guardian {
    private $flosc_state;
    private $rag_manager;

    public function __construct($flosc_state) {
        $this->flosc_state = $flosc_state;
        $this->rag_manager = new FLOSC_RAG_Manager();
    }

    public function flosc_execute_tool($tool_name, $args) {
        // CHECK ACCESS BEFORE EXECUTING
        $access_check = $this->flosc_check_tool_access($tool_name, $args);

        if (!$access_check['allowed']) {
            return $this->flosc_denial_payload($access_check['reason'], $access_check['cta']);
        }

        // Execute tool
        $result = $this->rag_manager->$tool_name($args);

        // Validate output
        return $this->flosc_validate_output($result, $tool_name);
    }

    private function flosc_check_tool_access($tool_name, $args) {
        $state = $this->flosc_state->flosc_get();

        switch ($tool_name) {
            case 'get_lesson_content':
                return $this->flosc_check_lesson_access($args['lesson_number'], $state);

            case 'search_posts':
            case 'search_knowledge_base':
                return ['allowed' => true];

            default:
                // Deny by default for unknown tools
                return ['allowed' => false, 'reason' => 'Unknown tool', 'cta' => null];
        }
    }

    private function flosc_check_lesson_access($lesson_number, $state) {
        $user_type = $state['user_type'];
        $access_level = $state['access_level'];

        // Admin: always allowed
        if ($user_type === 'admin') {
            return ['allowed' => true];
        }

        // Member: always allowed
        if ($access_level === 'member') {
            return ['allowed' => true];
        }

        // Guest: ONLY their free lesson
        if ($user_type === 'guest') {
            $free_lesson = $state['quiz']['free_lesson_number'];
            if ($lesson_number == $free_lesson) {
                return ['allowed' => true];
            }
            return [
                'allowed' => false,
                'reason' => 'Only your free lesson is available',
                'cta' => 'unlock_full_access'
            ];
        }

        // Visitor: MUST take quiz
        return [
            'allowed' => false,
            'reason' => 'Take quiz to unlock your free lesson',
            'cta' => 'take_quiz'
        ];
    }

    private function flosc_denial_payload($reason, $cta) {
        $messages = [
            'take_quiz' => "I'd love to show you that lesson! First, take our quick 2-minute quiz to get your personalized free lesson. 🎯",
            'unlock_full_access' => "Great choice! That lesson is part of full access. Unlock all lessons now to continue your learning journey. ✨",
            'login_required' => "Please log in to access your personalized content.",
        ];

        return [
            'denied' => true,
            'reason' => $reason,
            'cta' => $cta,
            'user_facing_message' => $messages[$cta] ?? "This content requires additional access."
        ];
    }

    private function flosc_validate_output($result, $tool_name) {
        if (isset($result['denied'])) return $result;

        // Log suspicious patterns
        if ($tool_name === 'get_lesson_content' && strlen($result['content'] ?? '') > 5000) {
            error_log("FLOSC Tool Guardian: Potential content leak detected");
        }

        return $result;
    }
}
```

The Plan Is Good — But It Misses the Elephant in the Room
Your plan correctly identifies the five things that need to happen. The task tables are precise. The verification steps are smart. But there's a fundamental architectural contradiction hiding under the surface that will make tomorrow painful if you don't address it first.
The Elephant: Two Chat Pipelines, Not One
The plan treats "the AI" as one system. It's actually two completely separate chat pipelines that don't talk to each other:
/chat (IVR pipeline)	/chat-rag (RAG pipeline)
Entry point	handle_chat() at flosc.php:3121	handle_chat_with_rag() at flosc.php:3276
Primary layer	IVR keyword matching	Anthropic tool-calling

The JS frontend only calls /chat. The RAG endpoint exists in PHP but nothing invokes it. Your plan's Requirement 5 (fully conversational) can't work through the IVR pipeline — the IVR is a keyword matcher that occasionally falls through to a 500-token AI call with no tool access.
You need to decide before tomorrow:
Option A: Merge RAG into the /chat pipeline. When the IVR doesn't match, instead of calling ai_factory->get_response() (no tools, 500 tokens), call the RAG handler (tools, 2000 tokens, conversation history). The IVR remains the first pass — but the fallback becomes the full AI, not a crippled one.
Option B: Switch the frontend to /chat-rag. Make the RAG pipeline primary and have it consult the IVR for scripted responses when appropriate (inverting the current relationship).
My recommendation: Option A. It preserves the IVR-first architecture you've built, respects the admin's ability to run without API keys, and is a surgical change — you're just swapping the fallback. Option B is a bigger rewrite and breaks the "zero API keys" IVR-only mode.

What's Missing from the Plan
1. The RAG system prompt is a second, divergent prompt builder.
build_rag_system_prompt() at flosc.php:3333 duplicates work from ai_factory->build_system_prompt() but with different content, hardcoded personality, and no knowledge file loading. IVR returns {content, user_autoprompts, phase_change}. AI returns a plain text string. The /chat handler has to normalize these. Your plan doesn't mention this — the merge needs to handle both return shapes.

The AI Provider Factory caches responses for 1 hour (set_transient).
